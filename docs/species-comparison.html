Overview

This script explores the alignment of the iNaturalist earthworm (Crassiclitellata) taxonomy with:

  • WoRMS (World Register of Marine Species), the current taxonomy source used by iNaturalist.

  • Misırlıoğlu et al.(2023), an updated global checklist of valid earthworm species.

Subspecies are excluded from this analysis.

Data cleaning

Simplify the raw data by retaining the following columns:

  • family

  • genus

  • specificEpithet

  • species_DATABASE

  • author (when available)

INAT

INAT <- INAT_raw %>% 
  filter(taxonRank == "species",
         order == "Crassiclitellata") %>% 
  select(family,
         genus,
         specificEpithet) %>% 
  mutate(species_INAT = paste(genus, specificEpithet)) %>% 
  arrange(family,
          genus,
          specificEpithet)

(at)thirty_legs commented in the iNaturalist flag:

“It also seems like iNaturalist did not import some species that have the uncertain/unassessed status on WoRMS, leading to further holes in the iNat taxonomy.”

Since the comment implies that only “some” species with these status were not imported,

I will only remove the “unaccepted” species from the database.

WORMS <- WORMS_raw %>%
  filter(RankName == "Species", 
         Status != "unaccepted") %>% 
  select(family = Family,
         genus = Genus,
         specificEpithet = Species,
         author = Authority,
         species_WORMS = ScientificName) %>% 
  distinct(species_WORMS,
           author,
           .keep_all = TRUE) %>%
  arrange(family,
          genus,
          specificEpithet)

Duplicate check:

WORMS_dup <- nrow(WORMS %>%
  count(species_WORMS) %>%
  filter(n > 1))

Data visualization

Coverage of iNaturalist earthworm species

coverage <- tibble(
  source = c("WoRMS", "WME"),
  matched = c(
    sppINAT - n_unmatchedWORMS,
    sppINAT - n_unmatchedWME
  ),
  unmatched = c(
    n_unmatchedWORMS,
    n_unmatchedWME
  )
)

coverage_long <- coverage %>%
  pivot_longer(
    c(matched, unmatched),
    names_to = "status",
    values_to = "species"
  ) %>%
  mutate(
    status = factor(status,
                    levels = c("unmatched", "matched"))
  )

fig_coverage <- ggplot(coverage_long,
       aes(source, species, fill = status)) +
  geom_col(
    width = 0.7
    ) +
  coord_flip() +
  scale_fill_manual(
    values = c(
      unmatched = "#4D4D4D",
      matched = "#74AC00"
    ),
    labels = c(
      unmatched = "Not matched",
      matched = "Matched"
    )
  ) +
  scale_y_continuous(
    expand = expansion(mult = c(0, 0.05))
  ) +
  labs(
    x = NULL,
    y = "iNaturalist earthworm species",
    fill = NULL
  ) +
  theme_classic(base_size = 12) +
  theme(
    legend.position = "top",
    legend.justification = "left",
    axis.title.y = element_blank(),
    axis.title.x = element_text(
      margin = margin(t = 10)
    ),
    axis.text = element_text(color = "black"),
    legend.text = element_text(size = 11)
  ) +
  geom_text(
  aes(label = ifelse(status == "unmatched",
                     paste0(round(species / sppINAT * 100, 1), "%"),
                     "")),
  position = position_stack(vjust = 0.5),
  color = "white",
  size = 3.5
)
ggsave(
  filename = here("outputs", "coverage_INAT.png"),
  plot = fig_coverage,
  width = 7,
  height = 5,
  units = "in",
  dpi = 600
)
